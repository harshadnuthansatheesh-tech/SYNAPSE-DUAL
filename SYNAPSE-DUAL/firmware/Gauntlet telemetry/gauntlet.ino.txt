#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>

Adafruit_MPU6050 mpu;

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22); // I2C SDA, SCL

  // Initialize MPU6050
  if (!mpu.begin()) {
    Serial.println("Failed to find MPU6050 chip");
    while (1) { delay(10); }
  }

  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
  
  Serial.println("Iron Man Gauntlet Telemetry Initialized.");
}

void loop() {
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  // Stream raw orientation data for gesture tracking / telemetry
  float pitch = a.acceleration.x;
  float roll = a.acceleration.y;

  // Simple gesture condition check
  if (pitch > 6.0) {
    Serial.println("GAUNTLET_GESTURE: REPULSOR_RAISED");
  }

  delay(20);
}
